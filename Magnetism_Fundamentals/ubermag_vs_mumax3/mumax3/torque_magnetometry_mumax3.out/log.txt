//output directory: torque_magnetometry_mumax3.out/
Ms0 := 9.65e5
Aex0 := 8.78e-13
Ku0 := 1.18e6
Hmag := 2.38e6
Bmag := mu0 * Hmag
SetGridSize(60, 60, 20)
SetCellSize(5e-9, 5e-9, 5e-9)
Msat = Ms0
Aex = Aex0
Ku1 = Ku0
anisU = vector(0, 0, 1)
B_ext = vector(0, 0, 0)
SnapshotFormat = "png"
m = Uniform(0, 0, 1)
Minimize()
SaveAs(m, "m_relaxed_zero_field.ovf")
SnapshotAs(m, "m_relaxed_zero_field.png")
print("<m> a campo cero:", m.Comp(0).Average(), m.Comp(1).Average(), m.Comp(2).Average())
//<m> a campo cero: 1.0099675920274522e-10 -4.2385525173611113e-10 0.9978158637152778
Fprintln("torque_vs_angle.txt", "# angulo_rad  tau_x  tau_y  tau_z   [N/m2 = J/m3]")
taumax := -1e30
taumin := 1e30
for k := 0.0; k < 71; k++ {
	phi := 2 * pi * k / 70
	By := Bmag * sin(phi)
	Bz := Bmag * cos(phi)
	B_ext = vector(0, By, Bz)
	Minimize()
	mx := m.Comp(0).Average()
	my := m.Comp(1).Average()
	mz := m.Comp(2).Average()
	tx := Ms0 * (my*Bz - mz*By)
	ty := Ms0 * (-mx * Bz)
	tz := Ms0 * (mx * By)
	Fprintln("torque_vs_angle.txt", phi, tx, ty, tz)
	taumax = max(taumax, tx)
	taumin = min(taumin, tx)
}
SaveAs(m, "m_last_angle.ovf")
print("tau_x maximo (N/m2):", taumax, " minimo:", taumin, " K =", Ku0)
